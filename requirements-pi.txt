# MH-FLOCKE — On-device (CPU-only) requirements
# ==============================================
# For running the brain on a low-power on-device target (e.g. Raspberry Pi /
# Pi Zero) with CPU-only PyTorch — no CUDA. Install with the CPU wheel index:
#
#   pip3 install -r requirements-pi.txt \
#     --index-url https://download.pytorch.org/whl/cpu \
#     --break-system-packages
#
# IMPORTANT: use --index-url for CPU-only PyTorch (~150 MB). Without it pip
# pulls the full CUDA build (~2 GB), which fails on small disks and has no GPU
# to use anyway.

numpy>=1.24.0
